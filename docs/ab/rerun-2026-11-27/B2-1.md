export interface SyncState {
  /** ISO-8601, UTC. */
  lastSyncedAt: string;
}

const INITIAL_STATE: SyncState = { lastSyncedAt: "1970-01-01T00:00:00.000Z" };

export function loadState(path: string): SyncState {
  if (!existsSync(path)) return { ...INITIAL_STATE };
  try {
    return JSON.parse(readFileSync(path, "utf8"));
  } catch {
    return { ...INITIAL_STATE };
  }
}

## Чому без `core` не вийде

Тип `Lead` лежить у `app/src/core/types.ts`, і поля для кампанії в ньому немає. Інтеграції отримують лід через контракт `Integration.send(lead: Lead)`. Обійти це через приведення типу чи локальне розширення `Lead & { utmCampaign?: string }` у `sheets-append.ts` означало б вигадати API поза ядром. До того ж нормалізація ліда могла б це поле просто відкинути.

## Що треба змінити в `core` (для PR платформної команди)

`app/src/core/types.ts`, в інтерфейсі `Lead`, після `budgetUsd?`:

```ts
  /** Рекламна кампанія з форми сайту (`utmCampaign`), якщо її передано. */
  utmCampaign?: string;
```

Поле необов'язкове, тому решта коду не зламається: Slack, Telegram і `sync` його просто не використовують. Після злиття треба оновити `scripts/core.lock.json`. Це теж робить платформна команда, бо `scripts/**` теж не редагується.

Ще одне: код, який перетворює заявку з форми на `Lead`, у цьому репозиторії відсутній. Вхід `runSync` уже має тип `Lead[]`. Тому `utmCampaign` треба передавати й там, де будується лід, інакше колонка завжди буде порожня.
